obs = obslua
local cur_settings
local output

function script_description()
    return "Used to modify gstreamer pipeline";
end

function button()
  local new_settings = obs.obs_data_create();
  obs.obs_data_set_string (new_settings, "preset", "speed");
  obs.obs_data_set_int (new_settings, "keyint_sec", 1);
  obs.obs_data_set_int (new_settings, "bitrate", 2500);
  obs.obs_data_set_string (new_settings, "profile", "main");
  obs.obs_data_set_string (new_settings, "ffmpeg_opts", "usage=ultralowlatency");

  local encoder = obs.obs_video_encoder_create("h264_texture_amf", "encoder_video", new_settings, NULL);
  obs.obs_encoder_set_video(encoder, obs.obs_get_video());
  
  output = obs.obs_output_create("gstreamer-output", "output-test", cur_settings, NULL);
  obs.obs_output_set_video_encoder(output, encoder);
  obs.obs_output_start(output);
return nil
end

function button2()
  obs.obs_output_stop(output);
return nil
end

function script_properties()
  local properties = obs.obs_properties_create()
  local textbox= obs.obs_properties_add_text (properties, "pipeline", "Pipeline", obs.OBS_TEXT_MULTILINE);
  local button = obs.obs_properties_add_button (properties, "new_button", "Start Pipeline", button)
  local button2 = obs.obs_properties_add_button (properties, "new_button2", "Stop Pipeline", button2)
return properties
end

function script_defaults(settings)
    obs.obs_data_set_default_string(settings, "pipeline", "autovideosink sync=false");
end

function script_load(settings)
    
end

function script_update(settings)
    cur_settings = settings;
end
